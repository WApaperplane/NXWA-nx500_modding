=== gen_menu.sh 尾部 ===
[ -z "$IP" ] && IP="WiFi未开"

# Telnet 状态
if ps -w 2>/dev/null | grep telnetd | grep -qv grep; then
    TN="Telnet开"
else
    TN="Telnet关"
fi

sed -e "s/%%IP%%/$IP/g" -e "s/%%TN%%/$TN/g" "$TPL" > "$DST.tmp" \
    && mv -f "$DST.tmp" "$DST"
exit 0
